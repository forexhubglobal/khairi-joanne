📊 Sambungan Google Sheets Pantas

Ingin setiap RSVP terus masuk ke Google Sheet anda secara langsung? Ikuti 3 langkah mudah ini:

  1. Buka Google Sheets baru di telefon atau laptop anda.
  2. Beri nama lajur: Nama, Telefon, Status, Pax, Ucapan, Tarikh.
  3. Klik menu Extensions > Apps Script, padam kod sedia ada dan tampal skrip di bawah:
function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < rows.length; i++) { var r = rows[i]; if (!r[0]) continue; var status = String(r[2] || "").trim().toUpperCase(); var isAttending = status.indexOf("HADIR") !== -1 && status.indexOf("TIDAK") === -1; var paxNum = parseInt(r[3], 10); result.push({ name: String(r[0] || "").trim(), phone: String(r[1] || "").trim(), attending: isAttending, pax: isNaN(paxNum) ? (isAttending ? 1 : 0) : paxNum, wish: String(r[4] || "").trim(), created_at: r[5] ? String(r[5]) : new Date().toISOString() }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); sheet.appendRow([ data.name, data.phone, data.attending ? "HADIR" : "TIDAK HADIR", data.pax, data.wish || "", new Date(data.created_at || new Date()).toLocaleString("ms-MY") ]); return ContentService.createTextOutput("OK"); }
